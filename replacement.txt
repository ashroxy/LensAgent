          (function() {
            try {
              if (window.__lensAgentPiiDirty === undefined) {
                window.__lensAgentPiiDirty = true;
                window.__lensAgentElements = [];
                const obs = new MutationObserver(() => { window.__lensAgentPiiDirty = true; });
                obs.observe(document, { childList: true, subtree: true, characterData: true, attributes: true });
                window.addEventListener('scroll', () => { window.__lensAgentPiiDirty = true; }, {passive: true, capture: true});
                window.addEventListener('resize', () => { window.__lensAgentPiiDirty = true; }, {passive: true});
              }

              if (!window.__lensAgentPiiDirty && window.__lensAgentElements) {
                return window.__lensAgentElements.map(item => {
                  const rect = item.el.getBoundingClientRect();
                  return {
                    id: item.data.id, type: item.data.type, category: item.data.category, redactionLabel: item.data.redactionLabel,
                    boundingBox: { x: Math.round(rect.left), y: Math.round(rect.top), width: Math.round(rect.width), height: Math.round(rect.height) }
                  };
                }).filter(i => i.boundingBox.width > 0 && i.boundingBox.height > 0);
              }

              const regexMap = [
                { cat: 'AADHAAR', rx: /(?<!\\d)[2-9]{1}[0-9]{3}[\\s\\-]?[0-9]{4}[\\s\\-]?[0-9]{4}(?!\\d)/ },
                { cat: 'PAN', rx: /\\b[A-Z]{5}[0-9]{4}[A-Z]{1}\\b/ },
                { cat: 'CREDITCARD', rx: /(?<!\\d)(?:\\d{4}[\\s\\-]?){3}\\d{4}(?!\\d)|(?<!\\d)\\d{15,16}(?!\\d)/ },
                { cat: 'INDIANPHONE', rx: /(?<!\\d)(?:\\+91[\\-\\s]?)?[6-9]\\d{9}(?!\\d)/ },
                { cat: 'EMAIL', rx: /\\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}\\b/ },
                { cat: 'UPIID', rx: /\\b[a-zA-Z0-9.\\-_]{2,256}@[a-zA-Z]{2,64}\\b/ },
                { cat: 'PASSPORT', rx: /\\b[A-Z]{1}[0-9]{7}\\b/ },
                { cat: 'DRIVINGLICENSE', rx: /\\b[A-Z]{2}[0-9]{2}[0-9]{11}\\b/ }
              ];
              const unifiedSource = regexMap.map(r => '(' + r.rx.source + ')').join('|');
              const unifiedRegex = new RegExp(unifiedSource, 'g');
              const sensitiveKeywords = /password|cvv|otp|pin|aadhaar|pan|card|secret|token|ssn|license/i;

              function getAllElementsDeep(root) {
                const elements = [];
                const walk = (node) => {
                  if (!node) return;
                  if (node.nodeType === Node.ELEMENT_NODE) { elements.push(node); if (node.shadowRoot) walk(node.shadowRoot); }
                  for (const child of node.children || []) walk(child);
                };
                walk(root); return elements;
              }

              const sensitiveItems = [];
              window.__lensAgentElements = [];
              let counter = 1;
              const allElements = getAllElementsDeep(document.body || document.documentElement);

              allElements.forEach((el) => {
                const tag = (el.tagName || '').toLowerCase();
                const className = (typeof el.className === 'string') ? el.className : '';
                const id = el.id || '';
                const dataPii = el.getAttribute ? el.getAttribute('data-pii') : null;
                const isSensitiveClass = /sensitive|pii-mask|mask-pii/i.test(className + ' ' + id);

                const pushItem = (cat, type, labelBase) => {
                  const data = { id: labelBase + '_' + counter, type, category: cat, redactionLabel: '[REDACTED_' + cat + '_#' + counter + ']' };
                  counter++;
                  window.__lensAgentElements.push({ el, data });
                  const rect = el.getBoundingClientRect();
                  if (rect.width > 0 && rect.height > 0) {
                    sensitiveItems.push({ ...data, boundingBox: { x: Math.round(rect.left), y: Math.round(rect.top), width: Math.round(rect.width), height: Math.round(rect.height) } });
                  }
                };

                if (dataPii || isSensitiveClass) {
                  pushItem(dataPii ? dataPii.toUpperCase() : 'SENSITIVE_CONTAINER', 'SENSITIVE_ELEMENT', 'SENSITIVE');
                  if (tag !== 'input' && tag !== 'textarea' && tag !== 'select') return;
                }

                if (tag === 'input' || tag === 'textarea' || tag === 'select' || el.isContentEditable) {
                  const val = (el.value || el.innerText || '').trim();
                  const type = el.type || 'text';
                  if (type === 'password') return pushItem('PASSWORD', 'INPUT_FIELD', 'SEC_PASSWORD');

                  const attrContext = ((el.name||'') + ' ' + (el.id||'') + ' ' + (el.placeholder||'') + ' ' + (el.getAttribute('aria-label')||'')).toLowerCase();
                  if (sensitiveKeywords.test(attrContext) || dataPii) {
                    let matchedCat = dataPii ? dataPii.toUpperCase() : 'AUTH_CREDENTIAL';
                    if (attrContext.includes('aadhaar')) matchedCat = 'AADHAAR';
                    else if (attrContext.includes('pan')) matchedCat = 'PAN';
                    else if (/cvv|pin|otp/.test(attrContext)) matchedCat = 'OTP_PIN';
                    return pushItem(matchedCat, 'INPUT_FIELD', 'SEC');
                  }

                  if (val.length > 0) {
                    unifiedRegex.lastIndex = 0;
                    const match = unifiedRegex.exec(val);
                    if (match) {
                      let matchedCat = 'PII_DATA';
                      for (let i = 1; i <= regexMap.length; i++) if (match[i] !== undefined) { matchedCat = regexMap[i-1].cat; break; }
                      return pushItem(matchedCat, 'INPUT_FIELD', 'SEC');
                    }
                  }
                }

                if (tag === 'img' || el.getAttribute('role') === 'img' || /avatar|profile|user-photo/i.test(el.className)) {
                  if (/avatar|profile|user|photo|badge/i.test((el.className||'') + ' ' + (el.id||'') + ' ' + (el.src||''))) {
                    pushItem('FACE_AVATAR', 'MEDIA_PII', 'FACE_AVATAR');
                  }
                }
              });

              window.__lensAgentPiiDirty = false;
              return sensitiveItems;
            } catch (e) {
              return [];
            }
          })()
